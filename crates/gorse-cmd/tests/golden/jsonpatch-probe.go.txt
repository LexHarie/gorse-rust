// The probe that recorded jsonpatch-go.txt from the Go module
// github.com/evanphx/json-patch/v5 v5.9.11 (the version go.mod pins).
// Run from inside the gorse module, then delete it:
//   go run ./zz_jpprobe < jsonpatch-cases.txt > jsonpatch-go.txt
package main

import (
	"bufio"
	"fmt"
	"os"

	jsonpatch "github.com/evanphx/json-patch/v5"
)

func main() {
	doc := []byte(`{"a":1,"arr":[1,2,3],"obj":{"x":"y"},"nul":null,"s":"str"}`)
	sc := bufio.NewScanner(os.Stdin)
	sc.Buffer(make([]byte, 1<<20), 1<<20)
	for sc.Scan() {
		line := sc.Text()
		p, err := jsonpatch.DecodePatch([]byte(line))
		if err != nil {
			fmt.Printf("%s\n  decode-err: %s\n", line, err.Error())
			continue
		}
		out, err := p.Apply(doc)
		if err != nil {
			fmt.Printf("%s\n  apply-err: %s\n", line, err.Error())
			continue
		}
		fmt.Printf("%s\n  ok: %s\n", line, string(out))
	}
}
